/**
 * ROSHDY BANK - Banking Management System
 * Developed by Roshdy Mohamed
 * Core concepts: Object-Oriented Programming, File Handling & Transactions
 */

#include <iostream>
#include <fstream>
#include <vector>
#include <iomanip>

class BankAccount {
private:
    int accountNumber;
    std::string ownerName;
    double balance;
    std::vector<std::string> transactionHistory;

public:
    BankAccount(int accNum, std::string name, double initialDep)
        : accountNumber(accNum), ownerName(name), balance(initialDep) {
        logTransaction("Account created with initial deposit: $" + std::to_string(initialDep));
    }

    void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            logTransaction("Deposited: $" + std::to_string(amount));
            saveToFile();
        }
    }

    bool withdraw(double amount) {
        if (amount > 0 && amount <= balance) {
            balance -= amount;
            logTransaction("Withdrawn: $" + std::to_string(amount));
            saveToFile();
            return true;
        }
        return false;
    }

    void saveToFile() {
        std::ofstream outFile("accounts.dat", std::ios::app);
        if (outFile.is_open()) {
            outFile << accountNumber << "|" << ownerName << "|" << balance << "\n";
            outFile.close();
        }
    }
};
Structured with separation of concerns and robust data persistence.
